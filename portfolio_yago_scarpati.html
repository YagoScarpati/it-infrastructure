Sou Yago Viana Scarpati, Especialista de Infraestrutura de TI com mais de 5 anos de experiência em ambientes corporativos de altíssima complexidade. PowerShell · Bulk-ADOperations.ps1 · exemplo
# Operações em massa via CSV — exemplo de desativação em lote # CSV esperado: SAMAccount, Reason, RequestedBy param ( [Parameter(Mandatory)] [string] $CsvPath, [switch] $WhatIf ) $users = Import-Csv $CsvPath $results = [System.Collections.Generic.List[PSObject]]::new() foreach ($u in $users) { try { $adUser = Get-ADUser $u.SAMAccount -Properties LastLogonDate if (-not $WhatIf) { Disable-ADAccount -Identity $u.SAMAccount Move-ADObject -Identity $adUser.DistinguishedName ` -TargetPath "OU=Disabled,DC=contoso,DC=com" } $results.Add([PSCustomObject]@{ SAM = $u.SAMAccount Status = if ($WhatIf) { "SIMULADO" } else { "DESATIVADO" } Motivo = $u.Reason Data = Get-Date -Format "dd/MM/yyyy HH:mm" }) } catch { Write-Warning "Erro em $($u.SAMAccount): $_" } } $results | Export-Csv ".\bulk_result_$(Get-Date -Format 'yyyyMMdd').csv" -NoTypeInformation Write-Host "✅ Processados: $($results.Count) usuários" -ForegroundColor Green
